    <script>
        const ALL_PLAYERS = [
            { id: 1, name: "Luka Dončić", team: "DAL", pos: "BC", price: 15.5, opponent: "@BOS" },
            { id: 2, name: "Stephen Curry", team: "GSW", pos: "BC", price: 14.0, opponent: "vs LAL" },
            { id: 3, name: "Shai Gilgeous-Alexander", team: "OKC", pos: "BC", price: 14.8, opponent: "@PHX" },
            { id: 4, name: "Ja Morant", team: "MEM", pos: "BC", price: 12.5, opponent: "vs HOU" },
            { id: 5, name: "Nikola Jokić", team: "DEN", pos: "FC", price: 16.0, opponent: "@MIN" },
            { id: 6, name: "Giannis Antetokounmpo", team: "MIL", pos: "FC", price: 15.2, opponent: "vs MIA" },
            { id: 7, name: "Jayson Tatum", team: "BOS", pos: "FC", price: 13.8, opponent: "@NYK" },
            { id: 8, name: "Victor Wembanyama", team: "SAS", pos: "FC", price: 13.0, opponent: "vs LAC" }
        ];

        let currentGW = 1;
        let rosterData = {};

        function initGWData(gw) {
            if (!rosterData[gw]) {
                if (gw > 1 && rosterData[gw - 1]) {
                    rosterData[gw] = JSON.parse(JSON.stringify(rosterData[gw - 1]));
                } else {
                    rosterData[gw] = {
                        BC: [null, null, null],
                        FC: [null, null, null]
                    };
                }
            }
        }

        function toggleTheme() {
            const body = document.body;
            const btn = document.getElementById('theme-toggle');
            if (body.getAttribute('data-theme') === 'dark') {
                body.setAttribute('data-theme', 'light');
                btn.innerText = '🌙 Tryb Ciemny';
            } else {
                body.setAttribute('data-theme', 'dark');
                btn.innerText = '☀️ Tryb Jasny';
            }
        }

        function changeGW(delta) {
            if (currentGW + delta < 1) return;
            currentGW += delta;
            document.getElementById('current-gw').innerText = `GW ${currentGW}`;
            initGWData(currentGW);
            renderGrids();
        }

        function filterPlayers() {
            const query = document.getElementById('search-input').value.toLowerCase();
            const formation = document.getElementById('filter-formation').value;
            const maxPrice = document.getElementById('filter-price').value;
            const sortBy = document.getElementById('sort-by').value;

            let result = ALL_PLAYERS.filter(p => {
                const matchesQuery = p.name.toLowerCase().includes(query) || p.team.toLowerCase().includes(query);
                const matchesPos = formation === 'ALL' || p.pos === formation;
                const matchesPrice = maxPrice === 'ALL' || p.price <= parseFloat(maxPrice);
                return matchesQuery && matchesPos && matchesPrice;
            });

            if (sortBy === 'name') result.sort((a,b) => a.name.localeCompare(b.name));
            if (sortBy === 'price-desc') result.sort((a,b) => b.price - a.price);
            if (sortBy === 'price-asc') result.sort((a,b) => a.price - b.price);

            renderSearchResults(result);
        }

        function renderSearchResults(players) {
            const container = document.getElementById('search-results');
            container.innerHTML = '';
            players.forEach(player => {
                const chip = document.createElement('div');
                chip.className = 'player-chip';
                chip.innerHTML = `
                    <span class="pos-tag ${player.pos}">${player.pos}</span>
                    <strong>${player.name}</strong> (${player.team}) - $${player.price}M
                `;
                chip.onclick = () => addPlayerToRoster(player);
                container.appendChild(chip);
            });
        }

        function addPlayerToRoster(player) {
            const currentRoster = rosterData[currentGW][player.pos];
            const emptyIndex = currentRoster.findIndex(slot => slot === null);

            if (emptyIndex === -1) {
                alert(`Brak wolnych miejsc w formacji ${player.pos}!`);
                return;
            }

            const exists = [...rosterData[currentGW].BC, ...rosterData[currentGW].FC]
                .some(p => p && p.id === player.id);
            if (exists) {
                alert("Ten zawodnik znajduje się już w Twoim składzie!");
                return;
            }

            currentRoster[emptyIndex] = player;
            renderGrids();
        }

        function removePlayer(pos, index) {
            rosterData[currentGW][pos][index] = null;
            renderGrids();
        }

        function renderFormGrid(posKey, elementId) {
            const container = document.getElementById(elementId);
            container.innerHTML = '';

            const cornerHeader = document.createElement('div');
            cornerHeader.className = 'sheet-cell sheet-header-cell';
            cornerHeader.innerText = `Zawodnik (${posKey})`;
            container.appendChild(cornerHeader);

            for (let i = 1; i <= 7; i++) {
                const gdHeader = document.createElement('div');
                gdHeader.className = 'sheet-cell sheet-header-cell';
                gdHeader.innerText = `GD ${i}`;
                container.appendChild(gdHeader);
            }

            const slots = rosterData[currentGW][posKey];
            slots.forEach((player, index) => {
                const labelCell = document.createElement('div');
                labelCell.className = 'sheet-cell player-label-cell';
                if (player) {
                    labelCell.innerHTML = `
                        <div>
                            <div>${player.name}</div>
                            <small style="color:var(--text-sub); font-size: 0.75rem">${player.team} | $${player.price}M</small>
                        </div>
                    `;
                } else {
                    labelCell.innerHTML = `<span style="color:var(--text-sub)">+ Wolny Slot (${posKey})</span>`;
                }
                container.appendChild(labelCell);

                for (let gd = 1; gd <= 7; gd++) {
                    const slotCell = document.createElement('div');
                    slotCell.className = 'sheet-cell';
                    
                    if (player) {
                        slotCell.innerHTML = `
                            <div class="slot-card">
                                <span class="remove-btn" onclick="removePlayer('${posKey}', ${index})">✕</span>
                                <div>Mecz</div>
                                <strong style="color:var(--brand-accent)">${player.opponent}</strong>
                            </div>
                        `;
                    } else {
                        slotCell.innerHTML = `<div class="slot-card empty">-</div>`;
                    }
                    container.appendChild(slotCell);
                }
            });
        }

        function renderGrids() {
            renderFormGrid('BC', 'grid-bc');
            renderFormGrid('FC', 'grid-fc');
        }

        initGWData(1);
        filterPlayers();
        renderGrids();
    </script>
